# What the workbench and the converter need from pip. start-workbench.bat /

# is missing every time they start; nothing is reinstalled once present.
# By hand: python -m pip install -r requirements.txt
#
# Each line ends with "# import <module>": the launcher checks that import
# to know a package is there, without asking pip.

openpyxl        # import openpyxl     -- Excel: DataSource workbooks (converter), uploaded test cases
pypdf           # import pypdf        -- a PDF spec or story attachment
pyflakes        # import pyflakes     -- the workbench's own code checks
